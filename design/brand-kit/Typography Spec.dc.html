@import url('https://fonts.googleapis.com/css2?family=Roboto:wght@400;600;700&family=Source+Serif+4:wght@700&display=swap');

:root {
  --font-heading: "Source Serif 4", serif;
  --font-body: Roboto, sans-serif;
  --navy: #262A78;  --ink: #0F1130;  --green: #55B847;
  --green-dark: #2F7A28;  --text: #45475A;  --cream: #F4F2EC;
}
body      { font: 400 16px/26px var(--font-body); color: var(--text); }
h1        { font: 700 clamp(32px, 3.6vw, 54px)/1.1 var(--font-heading); letter-spacing: -0.01em; }
h2        { font: 700 clamp(28px, 3vw, 44px)/1.12 var(--font-heading); letter-spacing: -0.005em; color: var(--navy); }
